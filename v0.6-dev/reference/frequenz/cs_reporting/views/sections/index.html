    def render_data_export(
    master_df: pd.DataFrame,
    mapper: ColumnMapper,
    tables_dict: TablesResult | None = None,
    resolution: Any = None,
) -> None:
    """Render predefined and custom CSV exports for Reporting and Battery Reporting."""
    inject_style("data_export.css")
    datasets = _available_export_datasets(master_df, mapper, tables_dict, resolution)

    st.markdown("### Netzbetreiber Daten & DA Preise")
    tables.aggrid_table(
        _round_numeric_columns(_build_dso_price_data_df(master_df)),
        key_prefix="data_export_dso_price",
        download_file_name="netzbetreiber_day_ahead_preis.csv",
    )

    st.divider()
    st.markdown("### Export-Vorlagen")
    export_tabs = st.tabs(list(datasets), key="data_export_standard_tabs")
    for index, (tab, (dataset_name, dataset_df)) in enumerate(
        zip(export_tabs, datasets.items())
    ):
        with tab:
            tables.aggrid_table(
                _round_numeric_columns(dataset_df),
                key_prefix=f"data_export_standard_{index}",
                download_file_name=f"{dataset_name.lower().replace(' ', '_')}.csv",
            )

    st.divider()
    st.markdown("### Individueller Datenexport")
    full_df = datasets["Gesamt-Datensatz"]
    default_columns = ["Zeitpunkt"] if "Zeitpunkt" in full_df.columns else []
    tables.aggrid_table(
        _round_numeric_columns(full_df),
        key_prefix="data_export_custom",
        download_file_name="individueller_datenexport.csv",
        column_selection=True,
        initially_visible_columns=default_columns,
    )

    def render_data_tabs(
    master_df: pd.DataFrame,
    tables_dict: TablesResult,
    mapper: ColumnMapper,
    mcfg: Any,
    component_types: Iterable[str],
    *,
    stacked: bool = False,
) -> None:
    """Render data tables for overview and component analyses.

    Args:
        master_df: Processed master dataframe.
        tables_dict: Mapping of table names to dataframes for each analysis.
        mapper: Column mapper for standardizing Gesamt-Datensatz column names.
        mcfg: Microgrid configuration object containing component metadata.
        component_types: Component type identifiers present in the microgrid.
        stacked: Render each dataset beneath the preceding one instead of in tabs.

    Returns:
        Streamlit components are rendered directly.
    """
    available_specs = []
    component_source_frames_by_table: dict[str, dict[str, pd.DataFrame]] = {}
    for spec in TABLE_TAB_SPECS:
        analysis_key = spec.get("analysis_key")
        if analysis_key in PLOT_SOURCE_ANALYSIS_KEYS:
            table_key = str(spec["table_key"])
            source_frames = {
                source: _component_analysis_table_for_source(
                    master_df,
                    mcfg,
                    component_types,
                    str(analysis_key),
                    source,
                )
                for source in PLOT_SOURCE_OPTIONS
            }
            if any(not frame.empty for frame in source_frames.values()):
                available_specs.append(spec)
                component_source_frames_by_table[table_key] = source_frames
        elif _table_spec_has_data(spec, tables_dict, master_df, mcfg, component_types):
            available_specs.append(spec)

    include_master = master_df is not None and not master_df.empty
    if not available_specs and not include_master:
        st.info("Keine Datentabellen verfügbar.")
        return

    def render_spec(spec: dict[str, Any]) -> None:
        """Render a single predefined data table."""
        value = tables_dict.get(spec["table_key"])
        analysis_key = spec.get("analysis_key")
        empty_info = spec.get("empty_info")
        if analysis_key in PLOT_SOURCE_ANALYSIS_KEYS:
            table_key = str(spec["table_key"])
            analysis_key = str(analysis_key)
            source_frames = component_source_frames_by_table[table_key]
            selected_source = next(
                source
                for source in PLOT_SOURCE_OPTIONS
                if not source_frames[source].empty
            )
            render_component_source_table(
                source_frames,
                key_prefix=spec["key_prefix"],
                default_source=selected_source,
            )
        else:
            render_table_section(
                value if isinstance(value, pd.DataFrame) else None,
                key_prefix=spec["key_prefix"],
                empty_info=empty_info,
            )

    if stacked:
        if include_master:
            st.subheader("Kerndaten")
            render_core_data(master_df, mapper)
            st.subheader("Gesamt-Datensatz")
            render_master_df(master_df, mapper)
        for spec in available_specs:
            st.subheader(str(spec["label"]))
            render_spec(spec)
        return

    tab_labels: list[str] = []
    if include_master:
        tab_labels.extend(("Kerndaten", "Gesamt-Datensatz"))
    tab_labels.extend(spec["label"] for spec in available_specs)
    tabs = st.tabs(tab_labels, key="reporting_data_tabs", on_change="ignore")

    table_tabs = tabs
    if include_master:
        with tabs[0]:
            render_core_data(master_df, mapper)
        with tabs[1]:
            render_master_df(master_df, mapper)
        table_tabs = tabs[2:]

    for tab, spec in zip(table_tabs, available_specs):
        with tab:
            render_spec(spec)

    def render_energy_pie_chart(
    power_df: pd.DataFrame | None,
    color_dict: dict[str, str] | None = None,
) -> None:
    """Render the energy mix pie chart from a power dataframe.

    Args:
        power_df: Dataframe containing ``Energy Source`` and ``Energy [kWh]``.
        color_dict: Optional color mapping for pie segments.

    Returns:
        Streamlit components are rendered directly.
    """
    if power_df is None or power_df.empty:
        st.info("Keine Daten für das Energie-Kreisdiagramm verfügbar.")
        return

    required_cols = {"Energy Source", "Energy [kWh]"}
    if not required_cols.issubset(set(power_df.columns)):
        st.info("Energiedaten enthalten nicht alle erforderlichen Spalten.")
        return

    power_df["Energy Source"] = power_df["Energy Source"].replace(
        {"CHP": "KWK", "Grid Consumption": "Netzbezug"}
    )
    power_df = power_df.rename(
        columns={"Energy Source": "Energiebezug", "Energy [kWh]": "Energie [kWh]"}
    )
    fig = plot_energy_pie_chart(power_df, color_dict=color_dict)
    render_plot_card("Energie-Mix", fig)

    def render_plots_tabs(
    tables: TablesResult,
    mapper: ColumnMapper,
    component_types: Iterable[str],
    master_df: pd.DataFrame,
    mcfg: Any,
    color_dict: dict[str, str] | None = None,

    """
    Render the plot tabs.

    Args:
        tables: The tables result containing data for the plots.
        mapper: The column mapper for renaming columns.
        component_types: Component type identifiers present in the microgrid.
        master_df: Master dataframe used to rebuild filtered plot analyses.
        mcfg: Microgrid configuration object containing component metadata.

    """

    tab_labels = [label for label, _ in plot_tabs_config]
    for tab, (_, render_fn) in zip(st.tabs(tab_labels), plot_tabs_config):
        with tab:
            render_fn()

                    )

                for spec, group in zip(box_specs, groups)

    title: str = "Zeitreihen-Plot",

    dotted_cols: list[str] | None = None,
    header_control: Callable[[], None] | None = None,
    key: str | None = None,
) -> None:
    """Render a generic time-series plot inside a card.

    Args:
        df: Dataframe containing a datetime column and series to plot.
        time_col: Column name containing datetime values.
        cols: Optional list of series columns to include; defaults to all
            non-time columns.
        title: Title displayed above the plot.
        xaxis_title: X-axis label.
        yaxis_title: Y-axis label.
        legend_title: Legend title or ``None`` to hide the legend title.
        color_dict: Optional mapping from column names to colors.
        long_format_flag: Whether the dataframe is already in long format.
        category_col: Category column name when ``long_format_flag`` is ``True``.
        value_col: Value column name when ``long_format_flag`` is ``True``.
        fill_cols: Columns to fill under the curve for stacked plots.
        plot_order: Explicit ordering of series when rendering.
        dotted_cols: Columns to render with dotted lines.
        header_control: Optional Streamlit control rendered at the right side of
            the plot card header.
        key: Optional key used to style plot cards with header controls.

    Returns:
        Streamlit components are rendered directly.
    """
    if df is None or df.empty:
        st.info("Keine Daten zum Plotten verfügbar.")
        return

    if time_col not in df.columns:
        st.info(f"Keine gültige Zeitspalte gefunden (erwartet: '{time_col}').")
        return

    df = df.copy()
    if not pd.api.types.is_datetime64_any_dtype(df[time_col]):
        df[time_col] = pd.to_datetime(df[time_col], errors="coerce")

    fig = plot_time_series(
        df,
        time_col=time_col,
        cols=cols,
        title="",
        xaxis_title=xaxis_title,
        yaxis_title=yaxis_title,
        legend_title=legend_title,
        color_dict=color_dict,
        long_format_flag=long_format_flag,
        category_col=category_col,
        value_col=value_col,
        fill_cols=fill_cols,
        dotted_cols=dotted_cols,
        plot_order=plot_order,
    )
    _apply_compact_time_series_layout(fig)
    _left_align_plot_title(fig)

    render_plot_card(title, fig, header_control=header_control, key=key)
